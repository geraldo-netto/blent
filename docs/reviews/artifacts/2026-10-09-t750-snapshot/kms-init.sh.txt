#!/bin/sh
[ "$$" -eq 1 ] || exit 64
export PATH=/usr/bin:/bin:/usr/sbin
mount -t proc proc /proc
mount -t sysfs sysfs /sys
mount -t devtmpfs devtmpfs /dev
mkdir -p /dev/pts /run/udev /tmp/.X11-unix
mount -t devpts devpts /dev/pts
/usr/lib/systemd/systemd-udevd --daemon
/bin/modprobe virtio_pci
/bin/modprobe virtio_gpu
/bin/modprobe 9pnet_virtio
/bin/modprobe 9p
/bin/modprobe evdi initial_device_count=2
sleep 1
case " $(cat /proc/cmdline) " in *" blent-variant=fragmented "*) export SNAP_FRAGMENTED=1;; esac
/bin/busybox mkfifo /tmp/video
LD_PRELOAD=/observe.so /helper --fps 60 --edid /edid.bin --capture-fifo /tmp/video > /tmp/helper.log 2>&1 &
helper=$!
/consume > /tmp/consumer.log 2>&1 &
consumer=$!
sleep 1
/produce > /tmp/producer.log 2>&1 &
producer=$!
wait "$consumer"
consumer_status=$?
wait "$producer"
producer_status=$?
kill "$helper"
wait "$helper"
cat /tmp/producer.log /tmp/helper.log /tmp/consumer.log
printf 'SNAP_STATUS consumer=%s producer=%s\n' "$consumer_status" "$producer_status"
dmesg | tail -35
poweroff -f
