import subprocess,json,itertools,pathlib,statistics
base=pathlib.Path('/tmp/blent-next-20261009')
rows=[]
for trial in range(5):
 for scale,workers,kind,cursor in itertools.product([1,2,4],[1,4],range(3),range(2)):
  for canonical in ([0,1] if trial%2 else [1,0]):
   result=subprocess.run([str(base/'t783-bench'),*map(str,[scale,workers,canonical,kind,cursor])],capture_output=True,text=True,check=True)
   record=json.loads(next(line[6:] for line in result.stdout.splitlines() if line.startswith('BENCH ')))
   rows.append(dict(record,trial=trial))
 (base/'t783-paired.json').write_text(json.dumps(rows,indent=2)+'\n')
 print('trial',trial,'done',flush=True)
for scale,workers,kind,cursor in itertools.product([1,2,4],[1,4],range(3),range(2)):
 if kind==2 or cursor==0: continue
 group=[r for r in rows if (r['scale'],r['workers'],r['kind'],r['cursor'])==(scale,workers,kind,cursor)]
 print(scale,workers,kind,cursor,[(canonical,statistics.median(r['wall_p50_ns'] for r in group if r['canonical']==canonical)/1000,statistics.median(r['cpu_ns_per_frame'] for r in group if r['canonical']==canonical)/1000) for canonical in [1,0]])
