base=/backups/disk2/blent-work/2026-10-09-evdi-second

sdk=/var/lib/flatpak/runtime/org.gnome.Sdk/x86_64/51/9283544305146d753f620e8b92976537033a439a942687c7c601e4b0a12a2df9/files
gl=/home/netto/.local/share/flatpak/runtime/org.freedesktop.Platform.GL.default/x86_64/26.08/badc1097c419e73781cf098413ffa8e7bf7db3528572c11712265b9193921842/files
